  const [yearFilter, setYearFilter] = useState(''); 
  const [monthFilters, setMonthFilters] = useState([]); 
  
  const [showFilterSheet, setShowFilterSheet] = useState(false);
  const [showProfileSideBar, setShowProfileSideBar] = useState(false); 
  const [isExporting, setIsExporting] = useState(false);

  const [addMainType, setAddMainType] = useState('ticket'); 
  const [addSubType, setAddSubType] = useState('flight');
  
  const [addForm, setAddForm] = useState({ title: '', from: '', to: '', rating: 0, apiImage: null });
  const [localPreviewImages, setLocalPreviewImages] = useState([]);
  
  // 新增：用于智能补全的焦点状态控制
  const [focusedInput, setFocusedInput] = useState(null); // 'from' 或 'to' 或 null

  const fileInputRef = useRef(null);

  const filteredAndSortedData = useMemo(() => {
    let result = [...initialLogData];
    if (activeFilters.length > 0) result = result.filter(item => activeFilters.includes(item.subType));
    
    if (yearFilter) {
      result = result.filter(item => item.date.startsWith(yearFilter));
    }
    if (monthFilters.length > 0) {
      result = result.filter(item => {
        const itemMonth = item.date.split('-')[1]; 
        return monthFilters.includes(itemMonth);
      });
    }
    
    result.sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
    return result;
  }, [activeFilters, yearFilter, monthFilters]);

  const handleSmartSearch = () => {
    if (!addForm.title) return;
    setTimeout(() => {
      setAddForm(prev => ({ 
        ...prev, 
        apiImage: 'https://images.unsplash.com/photo-1578374173705-969cbe6f2d6b?auto=format&fit=crop&w=200&q=80' 
      }));
    }, 600);
  };

  const handleImageSelect = (e) => {
    const files = Array.from(e.target.files);
    const newImages = files.map(file => URL.createObjectURL(file));
    setLocalPreviewImages(prev => [...prev, ...newImages]);
  };

  const resetAddForm = () => {
    setAddForm({ title: '', from: '', to: '', rating: 0, apiImage: null });
    setLocalPreviewImages([]);
    setFocusedInput(null);
  }

  const selectSuggestion = (field, item) => {
    setAddForm(prev => ({ ...prev, [field]: item.name }));
    setFocusedInput(null);
  };

  const renderListItem = (item, index) => {
    let icon, colorClass;
    if (item.baseType === 'ticket') {
      icon = item.subType === 'flight' ? <Plane size={20} /> : (item.subType === 'train' ? <Train size={20} /> : <MapIcon size={20}/>);
      colorClass = item.subType === 'flight' ? 'bg-blue-100 text-[#007AFF]' : 'bg-orange-100 text-orange-500';
    } else if (item.baseType === 'location') {
      icon = <MapPin size={20} />;
      colorClass = 'bg-red-100 text-red-500';
    } else {
      icon = item.subType === 'game' ? <Gamepad2 size={20} /> : (item.subType === 'tv' ? <Tv size={20} /> : (item.subType === 'book' ? <Book size={20} /> : <Film size={20} />));
      colorClass = 'bg-purple-100 text-purple-500';
    }

    const isLast = index === filteredAndSortedData.length - 1;

    return (
      <div key={item.id} className={`p-4 flex gap-4 ${!isLast ? 'border-b border-gray-100' : ''}`}>
        {item.apiImage ? (
          <div className="w-14 h-20 shrink-0 rounded-md overflow-hidden shadow-sm border border-gray-200 bg-gray-100">
            <img src={item.apiImage} alt="cover" className="w-full h-full object-cover" />
          </div>
        ) : (
          <div className={`w-12 h-12 rounded-xl flex items-center justify-center shrink-0 ${colorClass}`}>{icon}</div>
        )}

        <div className="flex-1 min-w-0">
          <div className="flex justify-between items-start mb-1">
            <h3 className="font-semibold text-[16px] text-gray-900 truncate">{item.title}</h3>
            {/* 隐藏年份前缀让显示更简洁 */}
            <span className="text-[12px] text-gray-400 shrink-0 ml-2">{item.date.replace(/^\d{4}-/, '')}</span>
          </div>
          <Stars count={item.rating} />
          {item.subtitle && <p className="text-[14px] text-gray-500 leading-snug mt-1 line-clamp-2">{item.subtitle}</p>}
          
          {item.localImages && item.localImages.length > 0 && (
            <div className="mt-3 flex gap-2 overflow-x-auto pb-1 scrollbar-hide">
              {item.localImages.map((imgUrl, i) => (
                <div key={i} className="w-20 h-20 shrink-0 rounded-lg overflow-hidden border border-gray-200">
                  <img src={imgUrl} alt="local" className="w-full h-full object-cover" />
                </div>
              ))}
            </div>
          )}
        </div>
      </div>
    );
  };

  return (
    <div className="bg-[#F2F2F7] min-h-screen font-sans text-gray-900 mx-auto max-w-md relative overflow-hidden shadow-2xl flex flex-col">
      {isExporting && <ExportVideoOverlay onComplete={() => setIsExporting(false)} />}
      
      <div className="h-12 flex justify-between items-center px-6 pt-2 text-[14px] font-medium shrink-0 bg-[#F2F2F7]/90 backdrop-blur-md z-20">
        <span>9:41</span>
        <div className="flex gap-1.5 items-center"><div className="w-4 h-3 bg-black rounded-sm"/><div className="w-4 h-3 bg-black rounded-sm"/><div className="w-6 h-3 border border-black rounded-sm relative"><div className="absolute inset-0.5 bg-black rounded-sm"/></div></div>
      </div>

      <div className="px-4 pt-3 pb-2 bg-[#F2F2F7]/90 backdrop-blur-md z-10 shrink-0">
        <div className="flex justify-between items-center mb-3">

          </button>
          <div className="flex bg-gray-200/80 p-0.5 rounded-lg mb-1">

            <button onClick={() => setViewMode('map')} className={`px-4 py-1.5 text-[13px] font-medium rounded-md transition-all ${viewMode === 'map' ? 'bg-white shadow-sm text-black' : 'text-gray-500'}`}>地图</button>
          </div>
        </div>
        {viewMode === 'list' && (
          <div className="flex gap-2 pb-1 mt-2 overflow-x-auto scrollbar-hide">
            <button onClick={() => {setActiveFilters([]); setYearFilter(''); setMonthFilters([]);}} className={`shrink-0 px-4 py-1.5 rounded-full text-[13px] font-medium transition-all ${(activeFilters.length === 0 && !yearFilter && monthFilters.length === 0) ? 'bg-black text-white' : 'bg-white text-gray-600 border border-gray-200 shadow-sm'}`}>全部</button>

'bg-blue-100 text-[#007AFF] border border-blue-200' : 'bg-white text-gray-600 border border-gray-200 shadow-sm'}`}>

          </div>

'bg-blue-50 text-[#007AFF] border-blue-200' : 'bg-gray-50 text-gray-600 border-gray-100'}`}>

                const isActive = activeFilters.includes(sub.id);
                return (
                  <button key={sub.id} onClick={() => { setActiveFilters(prev => prev.includes(sub.id) ? prev.filter(f => f !== sub.id) : [...prev, sub.id]) }} className={`px-4 py-2 rounded-full text-[14px] transition-all flex items-center gap-1 border ${isActive ? 'bg-purple-50 text-purple-600 border-purple-200' : 'bg-gray-50 text-gray-600 border-gray-100'}`}>
                    {isActive && <Check size={14} />} {sub.label}
                  </button>
                );
              })}
            </div>
          </div>
        </div>
      </div>

      {/* --- Add Modal --- */}
      <div className={`fixed inset-0 z-50 flex justify-center items-end bg-black/40 transition-opacity ${showAddModal ? 'opacity-100' : 'opacity-0 pointer-events-none'}`}>
        <div className={`w-full max-w-md bg-[#F2F2F7] rounded-t-3xl overflow-hidden transition-transform transform duration-300 flex flex-col ${showAddModal ? 'translate-y-0' : 'translate-y-full'}`} style={{ height: '90vh' }}>
          
          <div className="bg-white px-4 py-3 flex justify-between items-center border-b border-gray-200 shrink-0">
            <button onClick={() => setShowAddModal(false)} className="text-[#007AFF] text-[17px]">取消</button>
            <span className="font-semibold text-[17px]">记录</span>
            <button onClick={() => setShowAddModal(false)} className="text-[#007AFF] font-semibold text-[17px]">保存</button>
          </div>

          <div className="p-4 overflow-y-auto flex-1 pb-20">
            <div className="flex bg-gray-200/80 p-1 rounded-lg mb-4">
              {[{id: 'ticket', label: '旅票'}, {id: 'location', label: '地点'}, {id: 'media', label: '娱乐'}].map(type => (
                <button key={type.id} onClick={() => {setAddMainType(type.id); setAddSubType(type.id==='location'?'location':subTypeOptions[type.id][0].id);}} className={`flex-1 py-1.5 text-[14px] font-medium rounded-md transition-all ${addMainType === type.id ? 'bg-white shadow-sm text-black' : 'text-gray-500'}`}>{type.label}</button>
              ))}
            </div>

            {addMainType !== 'location' && (
              <div className="mb-4">
                <div className="flex flex-wrap gap-2">
                  {subTypeOptions[addMainType].map(sub => (
                    <button key={sub.id} onClick={() => setAddSubType(sub.id)} className={`px-4 py-1.5 rounded-full text-[13px] font-medium transition-all border ${addSubType === sub.id ? 'bg-[#007AFF] text-white border-[#007AFF]' : 'bg-white text-gray-600 border-gray-200 shadow-sm'}`}>{sub.label}</button>
                  ))}
                </div>
              </div>
            )}

            <div className="bg-white rounded-[10px] overflow-hidden shadow-sm border border-gray-100 mb-4">
              {(addMainType === 'media' || addMainType === 'location') && (
                <div className="flex border-b border-gray-100 p-3 items-center bg-blue-50/50">
                  <div className="flex-1 flex items-center bg-gray-100 rounded-lg px-3 py-1.5">
                    <Search size={16} className="text-gray-400 mr-2" />
                    <input 
                      type="text" 
                      className="flex-1 bg-transparent outline-none text-[15px]" 
                      value={addForm.title}
                      onChange={(e) => setAddForm({...addForm, title: e.target.value})}
                    />
                  </div>
                  <button onClick={handleSmartSearch} className="ml-3 text-[#007AFF] text-[14px] font-medium whitespace-nowrap">检索</button>
                </div>
              )}

              {addForm.apiImage && (
                <div className="p-4 border-b border-gray-100 flex justify-center bg-gray-50">
                  <div className="relative">
                    <img src={addForm.apiImage} alt="Cover Preview" className="h-32 rounded-lg shadow-md border border-gray-200" />
                    <button onClick={() => setAddForm({...addForm, apiImage: null})} className="absolute -top-2 -right-2 bg-gray-800 text-white rounded-full p-1 shadow"><X size={12} /></button>
                  </div>
                </div>
              )}

              {addMainType === 'ticket' && (
                <>
                  <div className="flex border-b border-gray-100 p-3.5 items-center">
                    <span className="w-16 text-[16px] text-gray-900 font-medium">班次</span>
                    <input type="text" className="flex-1 text-[16px] outline-none" value={addForm.title} onChange={e=>setAddForm({...addForm, title: e.target.value})} />
                  </div>
                  
                  {/* 出发地输入及联想 */}
                  <div className="relative">
                    <div className="flex border-b border-gray-100 p-3.5 items-center">
                      <span className="w-16 text-[16px] text-gray-900 font-medium">出发地</span>
                      <input 
                        type="text" 
                        className="flex-1 text-[16px] outline-none" 
                        value={addForm.from} 
                        onChange={e=>setAddForm({...addForm, from: e.target.value})}
                        onFocus={() => setFocusedInput('from')}
                        onBlur={() => setTimeout(() => setFocusedInput(null), 200)}
                      />
                    </div>
                    {focusedInput === 'from' && (addSubType === 'flight' || addSubType === 'train') && (
                      <div className="bg-gray-50 border-b border-gray-100 flex flex-col max-h-32 overflow-y-auto">
                        {(locationSuggestions[addSubType] || []).map((sug, i) => (
                          <div key={i} onClick={() => selectSuggestion('from', sug)} className="px-4 py-2.5 active:bg-gray-200 flex justify-between items-center cursor-pointer border-b border-gray-100 last:border-none">
                            <span className="text-[15px] text-gray-900">{sug.name}</span>
                            {sug.code && <span className="text-[12px] text-[#007AFF] bg-blue-100 px-1.5 py-0.5 rounded">{sug.code}</span>}
                          </div>
                        ))}
                      </div>
                    )}
                  </div>

                  {/* 目的地输入及联想 */}
                  <div className="relative">
                    <div className="flex border-b border-gray-100 p-3.5 items-center">
                      <span className="w-16 text-[16px] text-gray-900 font-medium">目的地</span>
                      <input 
                        type="text" 
                        className="flex-1 text-[16px] outline-none" 
                        value={addForm.to} 
                        onChange={e=>setAddForm({...addForm, to: e.target.value})}
                        onFocus={() => setFocusedInput('to')}
                        onBlur={() => setTimeout(() => setFocusedInput(null), 200)}
                      />
                    </div>
                    {focusedInput === 'to' && (addSubType === 'flight' || addSubType === 'train') && (
                      <div className="bg-gray-50 border-b border-gray-100 flex flex-col max-h-32 overflow-y-auto">
                        {(locationSuggestions[addSubType] || []).map((sug, i) => (
                          <div key={i} onClick={() => selectSuggestion('to', sug)} className="px-4 py-2.5 active:bg-gray-200 flex justify-between items-center cursor-pointer border-b border-gray-100 last:border-none">
                            <span className="text-[15px] text-gray-900">{sug.name}</span>
                            {sug.code && <span className="text-[12px] text-[#007AFF] bg-blue-100 px-1.5 py-0.5 rounded">{sug.code}</span>}
                          </div>
                        ))}
                      </div>
                    )}
                  </div>
                </>
              )}

              {(addMainType === 'location' || addMainType === 'media') && (
                <div className="flex border-b border-gray-100 p-3.5 items-center">
                  <span className="w-16 text-[16px] text-gray-900 font-medium">评分</span>
                  <Stars count={addForm.rating} onChange={(val) => setAddForm({...addForm, rating: val})} />
                </div>
              )}

              <div className="flex border-b border-gray-100 p-3.5 items-center">
                <span className="w-16 text-[16px] text-gray-900 font-medium">时间</span>
                {/* 动态切换日期选择器类型：票务使用 datetime-local */}
                <input 
                  type={addMainType === 'ticket' ? 'datetime-local' : 'date'} 
                  className="flex-1 text-[16px] outline-none text-gray-900 bg-transparent" 
                />
              </div>

              <div className="p-3.5">
                <textarea className="w-full text-[16px] outline-none resize-none h-24 text-gray-900"></textarea>
              </div>
            </div>

            <div className="bg-white rounded-[10px] p-4 shadow-sm border border-gray-100">
              <div className="flex justify-between items-center mb-3">
                <h3 className="text-[15px] font-medium flex items-center gap-2"><ImageIcon size={18} className="text-gray-400"/> 添加照片</h3>
                <input type="file" multiple accept="image/*" className="hidden" ref={fileInputRef} onChange={handleImageSelect} />
                <button onClick={() => fileInputRef.current?.click()} className="text-[#007AFF] bg-blue-50 px-3 py-1 rounded-full text-[13px] font-medium">选择</button>
              </div>

              {localPreviewImages.length > 0 && (
                <div className="flex gap-3 overflow-x-auto pb-2 scrollbar-hide">
                  {localPreviewImages.map((src, index) => (
                    <div key={index} className="w-20 h-20 shrink-0 relative rounded-lg overflow-hidden border border-gray-200 shadow-sm">
                      <img src={src} alt="local preview" className="w-full h-full object-cover" />
                      <button onClick={() => removeLocalImage(index)} className="absolute top-1 right-1 bg-black/50 backdrop-blur text-white rounded-full p-1"><X size={10} /></button>
                    </div>
                  ))}
                </div>
              )}
            </div>

          </div>
        </div>
      </div>

      {/* --- 侧边栏：我的 (Profile) - 限定为一半画幅 --- */}
      <div className={`fixed inset-0 z-50 transition-opacity duration-300 ${showProfileSideBar ? 'opacity-100' : 'opacity-0 pointer-events-none'}`}>
        {/* 右侧点击遮罩 */}
        <div className="absolute inset-0 bg-black/40" onClick={() => setShowProfileSideBar(false)}></div>
        {/* 左侧画幅面板 - 设置宽度为 w-1/2 (50%) */}
        <div className={`absolute top-0 left-0 w-1/2 h-full bg-[#F2F2F7] shadow-2xl transition-transform transform duration-300 flex flex-col ${showProfileSideBar ? 'translate-x-0' : '-translate-x-full'}`}>
           <ProfileView />
        </div>
      </div>

    </div>
  );
}
